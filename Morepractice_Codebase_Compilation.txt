import Foundation
import Combine
import FirebaseAuth
import FirebaseFirestore

class AppViewModel: ObservableObject {
    @Published var isAuthenticated = false
    @Published var currentUser: AppUser?
    @Published var isLoading = false
    
    private var cancellables = Set<AnyCancellable>()
    private let db = Firestore.firestore()
    
    init() {
        // Listen for auth state changes
        Auth.auth().addStateDidChangeListener { [weak self] (_, user) in
            DispatchQueue.main.async {
                self?.isAuthenticated = user != nil
                if let user = user {
                    self?.fetchUserData(uid: user.uid)
                } else {
                    self?.currentUser = nil
                }
            }
        }
    }
    
    private func fetchUserData(uid: String) {
        isLoading = true
        db.collection("users").document(uid).getDocument { [weak self] (document, error) in
            DispatchQueue.main.async {
                if let document = document, document.exists, let data = document.data() {
                    self?.currentUser = AppUser(
                        id: document.documentID,
                        name: data["name"] as? String ?? "",
                        email: data["email"] as? String ?? "",
                        isOnline: data["isOnline"] as? Bool ?? false,
                        uid: document.documentID,
                        similarityScore: 0.0
                    )
                }
                self?.isLoading = false
            }
        }
    }
}

import Foundation
import Combine
import FirebaseAuth
import FirebaseFirestore

class AuthViewModel: ObservableObject {
    @Published var email = ""
    @Published var password = ""
    @Published var name = ""
    @Published var errorMessage = ""
    @Published var isLoading = false
    @Published var isAuthenticated = false
    
    private let db = Firestore.firestore()
    private var cancellables = Set<AnyCancellable>()
    
    init() {
        // Listen for auth state changes
        Auth.auth().addStateDidChangeListener { [weak self] (_, user) in
            DispatchQueue.main.async {
                self?.isAuthenticated = user != nil
            }
        }
    }
    
    func signIn() {
        guard !email.isEmpty && !password.isEmpty else {
            errorMessage = "Please enter both email and password."
            return
        }
        
        isLoading = true
        errorMessage = ""
        
        Auth.auth().signIn(withEmail: email, password: password) { [weak self] (result, error) in
            DispatchQueue.main.async {
                self?.isLoading = false
                if let error = error {
                    self?.errorMessage = error.localizedDescription
                }
            }
        }
    }
    
    func signUp() {
        guard !email.isEmpty && !password.isEmpty && !name.isEmpty else {
            errorMessage = "Please fill in all fields."
            return
        }
        
        if password.count < 6 {
            errorMessage = "Password must be at least 6 characters long."
            return
        }
        
        isLoading = true
        errorMessage = ""
        
        Auth.auth().createUser(withEmail: email, password: password) { [weak self] (result, error) in
            if let error = error {
                DispatchQueue.main.async {
                    self?.isLoading = false
                    self?.errorMessage = error.localizedDescription
                }
                return
            }
            
            guard let userId = result?.user.uid else {
                DispatchQueue.main.async {
                    self?.isLoading = false
                    self?.errorMessage = "Failed to create user."
                }
                return
            }
            
            // Create user document in Firestore
            self?.createUserDocument(userId: userId) { success in
                DispatchQueue.main.async {
                    self?.isLoading = false
                    if !success {
                        self?.errorMessage = "Failed to create user profile."
                    }
                }
            }
        }
    }
    
    private func createUserDocument(userId: String, completion: @escaping (Bool) -> Void) {
        let userData: [String: Any] = [
            "name": name,
            "email": email,
            "isOnline": true,
            "createdAt": Timestamp(date: Date())
        ]
        
        db.collection("users").document(userId).setData(userData) { error in
            completion(error == nil)
        }
    }
    
    func signOut() {
        do {
            try Auth.auth().signOut()
        } catch {
            errorMessage = error.localizedDescription
        }
    }
}

//  ChatViewModel.swift

//  Created by Fred Olivier on 05/10/2024.

import Foundation
import FirebaseAuth
import FirebaseFirestore
import Combine

class ChatViewModel: ObservableObject {
    @Published var messages: [ChatMessage] = []
    @Published var newMessage = ""
    @Published var isLoading = false
    @Published var errorMessage = ""
    
    private let db = Firestore.firestore()
    private var chatId: String
    private var listener: ListenerRegistration?
    private var userId: String
    private var otherUserId: String
    
    init(chatId: String, userId: String, otherUserId: String) {
        self.chatId = chatId
        self.userId = userId
        self.otherUserId = otherUserId
        fetchMessages()
    }
    
    deinit {
        // Remove Firestore listener when ViewModel is deallocated
        listener?.remove()
    }
    
    func fetchMessages() {
        isLoading = true
        
        listener = db.collection("chats").document(chatId).collection("messages")
            .order(by: "timestamp", descending: false)
            .addSnapshotListener { [weak self] (snapshot, error) in
                guard let self = self else { return }
                
                DispatchQueue.main.async {
                    self.isLoading = false
                    
                    if let error = error {
                        self.errorMessage = error.localizedDescription
                        return
                    }
                    
                    guard let documents = snapshot?.documents else {
                        self.errorMessage = "No messages found."
                        return
                    }
                    
                    self.messages = documents.compactMap { document -> ChatMessage? in
                        let data = document.data()
                        guard let sender = data["sender"] as? String,
                              let content = data["content"] as? String,
                              let timestamp = data["timestamp"] as? Timestamp else {
                            return nil
                        }
                        
                        return ChatMessage(
                            id: document.documentID,
                            sender: sender,
                            content: content,
                            timestamp: timestamp.dateValue()
                        )
                    }
                }
            }
    }
    
    func sendMessage() {
        guard !newMessage.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else { return }
        
        let message: [String: Any] = [
            "sender": userId,
            "content": newMessage,
            "timestamp": Timestamp(date: Date())
        ]
        
        // Add message to the chat
        db.collection("chats").document(chatId).collection("messages").addDocument(data: message) { [weak self] error in
            DispatchQueue.main.async {
                if let error = error {
                    self?.errorMessage = error.localizedDescription
                } else {
                    self?.newMessage = ""
                    // Update the last message preview in the chat document
                    self?.updateLastMessagePreview()
                }
            }
        }
    }
    
    private func updateLastMessagePreview() {
        let lastMessagePreview = newMessage.prefix(30).appending(newMessage.count > 30 ? "..." : "")
        let chatData: [String: Any] = [
            "lastMessagePreview": lastMessagePreview,
            "lastUpdated": Timestamp(date: Date())
        ]
        
        db.collection("chats").document(chatId).setData(chatData, merge: true) { [weak self] error in
            if let error = error {
                DispatchQueue.main.async {
                    self?.errorMessage = error.localizedDescription
                }
            }
        }
    }
}

//
//  EphemeralChatViewModel.swift
//  Morepractice
//
//  Created by Fred Olivier on 10/10/2024.

import Foundation
import Combine
import FirebaseFirestore

class EphemeralChatViewModel: ObservableObject {
    @Published var messages: [ChatMessage] = []
    @Published var newMessage = ""
    @Published var isLoading = false
    @Published var errorMessage = ""
    
    private let db = Firestore.firestore()
    private var chatId: String
    private var userId: String
    private var otherUserId: String
    private var timer: Timer?
    private var messageExpirationTime: TimeInterval = 300 // 5 minutes by default
    
    init(chatId: String, userId: String, otherUserId: String, expirationTime: TimeInterval = 300) {
        self.chatId = chatId
        self.userId = userId
        self.otherUserId = otherUserId
        self.messageExpirationTime = expirationTime
        
        fetchMessages()
        startExpirationTimer()
    }
    
    deinit {
        timer?.invalidate()
    }
    
    private func startExpirationTimer() {
        timer = Timer.scheduledTimer(withTimeInterval: 60, repeats: true) { [weak self] _ in
            self?.checkAndRemoveExpiredMessages()
        }
    }
    
    private func checkAndRemoveExpiredMessages() {
        let currentTime = Date()
        
        messages = messages.filter { message in
            // Keep messages that haven't expired yet
            return currentTime.timeIntervalSince(message.timestamp) < messageExpirationTime
        }
    }
    
    func fetchMessages() {
        isLoading = true
        
        db.collection("ephemeralChats").document(chatId).collection("messages")
            .order(by: "timestamp", descending: false)
            .getDocuments { [weak self] (snapshot, error) in
                guard let self = self else { return }
                
                DispatchQueue.main.async {
                    self.isLoading = false
                    
                    if let error = error {
                        self.errorMessage = error.localizedDescription
                        return
                    }
                    
                    guard let documents = snapshot?.documents else {
                        return
                    }
                    
                    let currentTime = Date()
                    
                    self.messages = documents.compactMap { document -> ChatMessage? in
                        let data = document.data()
                        guard let sender = data["sender"] as? String,
                              let content = data["content"] as? String,
                              let timestamp = data["timestamp"] as? Timestamp else {
                            return nil
                        }
                        
                        let messageTimestamp = timestamp.dateValue()
                        
                        // Only include messages that haven't expired
                        if currentTime.timeIntervalSince(messageTimestamp) < self.messageExpirationTime {
                            return ChatMessage(
                                id: document.documentID,
                                sender: sender,
                                content: content,
                                timestamp: messageTimestamp
                            )
                        } else {
                            // Delete expired messages from Firestore
                            self.db.collection("ephemeralChats").document(self.chatId)
                                .collection("messages").document(document.documentID).delete()
                            return nil
                        }
                    }
                }
            }
    }
    
    func sendMessage() {
        guard !newMessage.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else { return }
        
        let message: [String: Any] = [
            "sender": userId,
            "content": newMessage,
            "timestamp": Timestamp(date: Date())
        ]
        
        // Add message to the chat
        db.collection("ephemeralChats").document(chatId).collection("messages").addDocument(data: message) { [weak self] error in
            DispatchQueue.main.async {
                if let error = error {
                    self?.errorMessage = error.localizedDescription
                } else {
                    self?.newMessage = ""
                    self?.fetchMessages() // Refresh messages after sending
                }
            }
        }
    }
}
